# jobpilot Kernel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the deterministic multi-agent kernel that every jobpilot agent plugs into: contracts, the handoff protocol, the capability broker, fail-closed guardrails, two-tier policy, a wiring compiler, the step executor, the supervisor with team breakers, a hash-chained audit log, contract drift detection and an eval runner. It is proven end to end by a two-agent demo pipeline.

**Architecture:** Agents are pure `run(input, ctx)` functions behind Zod contracts. A supervisor routes typed envelopes between them using a wiring graph compiled and checked at boot, and drains a SQLite work queue in a total order. Every step goes through one executor that checks guardrails, memoises by a content hash, enforces budgets and timeouts, and commits envelopes, step status and cost in one transaction. An LLM is only ever called inside an agent, through a broker-issued tool.

**Tech Stack:** Node 24 · TypeScript 5.9 (strict, NodeNext) · Zod 4.6 · better-sqlite3 13 · commander 15 · tsx 4 · vitest 5 · pnpm 10 (install only).

**Spec:** `docs/superpowers/specs/2026-10-08-jobpilot-architecture-design.md`: §0.3 two-tier policy, §5 contract model, §5.5 handoff protocol, §6 supervisor and state machine, §7.2 team leads, §8 guardrails, §11 data model, §14 testing, §15 evals.

**Verified:** every code block below was built and run in a scratch copy before this plan was written: 15 test files, 168 tests, typecheck clean, `node --run verify` green. (Hardened during execution after review: Task 3 contracts — governance fields in the hash, deep freeze, two-way capability and retry rules; Task 5 policy — host normalisation, own-property egress lookups, deep freeze; Task 6 broker — redirects, ports and credentials refused, provider failures mapped, one model call in flight, per-step spend cap; Task 7 guardrails — fail closed on any throw, misregistered mandatory checks, invalid verdicts and missing output; Task 8 wiring — one type id may not carry two schemas, the seed must be consumed; Task 9 job machine — QA retry bounded to one regeneration, NEEDS_HUMAN exits; Task 10 store — anchored audit-chain head with optional expected head, atomic version-guarded migrations, steps finish exactly once; Task 11 executor — input guardrails before replay, door quarantines not cached, spend summed across retries, events chained in the step transaction, never throws; broker refuses calls after a timeout; Task 12 supervisor — rejected transitions audited not fatal, open obligations and replay counts derived from run events, per-run breakers, run lifecycle guards; Task 13 drift check — stale exports, hand-edited shapes, newer versions, unreadable files and a missing directory are reported.)

`src/kernel/canonical.ts`:

```ts
import { createHash } from 'node:crypto';

/**
 * Deterministic JSON: object keys sorted by code unit, `undefined` object
 * properties dropped, `undefined` array items become null. Throws on values
 * that have no stable JSON form (NaN, Infinity, Date, Map, class instances).
 */
export function canonicalJson(value: unknown): string {
  return serialise(value, '$');
}

function serialise(v: unknown, path: string): string {
  if (v === null) return 'null';
  switch (typeof v) {
    case 'string':
      return JSON.stringify(v);
    case 'boolean':
      return v ? 'true' : 'false';
    case 'number':
      if (!Number.isFinite(v)) throw new TypeError(`canonicalJson: non-finite number at ${path}`);
      return JSON.stringify(Object.is(v, -0) ? 0 : v);
    case 'object': {
      if (Array.isArray(v)) {
        return `[${v.map((x, i) => serialise(x === undefined ? null : x, `${path}[${i}]`)).join(',')}]`;
      }
      const proto: unknown = Object.getPrototypeOf(v);
      if (proto !== Object.prototype && proto !== null) {
        throw new TypeError(`canonicalJson: unsupported object at ${path}`);
      }
      const rec = v as Record<string, unknown>;
      const parts: string[] = [];
      for (const k of Object.keys(rec).sort()) {
        const x = rec[k];
        if (x === undefined) continue;
        parts.push(`${JSON.stringify(k)}:${serialise(x, `${path}.${k}`)}`);
      }
      return `{${parts.join(',')}}`;
    }
    default:
      throw new TypeError(`canonicalJson: unsupported ${typeof v} at ${path}`);
  }
}

export function sha256(text: string): string {
  return createHash('sha256').update(text, 'utf8').digest('hex');
}

export function hashOf(value: unknown): string {
  return sha256(canonicalJson(value));
}
```

```ts
export const ERROR_CODES = [
  'rate_limit',
  'overloaded',
  'timeout',
  'network',
  'invalid_output',
  'policy_denied',
  'budget_exceeded',
  'unknown',
] as const;
export type ErrorCode = (typeof ERROR_CODES)[number];

/** Only these may appear in a contract's retry.on list. */
export const RETRYABLE: ReadonlySet<ErrorCode> = new Set(['rate_limit', 'overloaded', 'timeout', 'network']);

export class KernelError extends Error {
  constructor(
    readonly code: ErrorCode,
    message: string,
  ) {
    super(message);
    this.name = 'KernelError';
  }
}

export class WiringError extends Error {
  constructor(readonly problems: readonly string[]) {
    super(`pipeline wiring is invalid:\n- ${problems.join('\n- ')}`);
    this.name = 'WiringError';
  }
}

export class PolicyViolation extends Error {
  constructor(readonly violations: readonly string[]) {
    super(`user policy may only tighten product policy:\n- ${violations.join('\n- ')}`);
    this.name = 'PolicyViolation';
  }
}

export class InvalidTransition extends Error {
  constructor(machine: string, from: string, on: string) {
    super(`${machine}: no transition from ${from} on ${on}`);
    this.name = 'InvalidTransition';
  }
}

export function classify(err: unknown): ErrorCode {
  if (err instanceof KernelError) return err.code;
  if (typeof err === 'object' && err !== null && (err as { name?: unknown }).name === 'TimeoutError') return 'timeout';
  return 'unknown';
}
```

```ts
/**
 * P0 public job data · P1 contact PII · P2 offers and salary · P3 mailbox bodies.
 * Ordered by sensitivity for taint propagation: an output inherits the most
 * sensitive class among its inputs unless its agent is a declared redactor.
 */
export const DATA_CLASSES = ['P0', 'P1', 'P2', 'P3'] as const;
export type DataClass = (typeof DATA_CLASSES)[number];

export function maxClass(classes: readonly DataClass[]): DataClass {
  let max: DataClass = 'P0';
  for (const c of classes) if (DATA_CLASSES.indexOf(c) > DATA_CLASSES.indexOf(max)) max = c;
  return max;
}
```

```ts
import type { z } from 'zod';

export interface MessageType<T = unknown> {
  readonly name: string;
  readonly version: number;
  /** `Name@version`, the routing key. */
  readonly id: string;
  readonly schema: z.ZodType<T>;
}

export function defineType<T>(name: string, version: number, schema: z.ZodType<T>): MessageType<T> {
  if (!/^[A-Z][A-Za-z0-9]*$/.test(name)) throw new TypeError(`message type name must be PascalCase: ${name}`);
  if (!Number.isInteger(version) || version < 1) throw new TypeError(`message type version must be a positive integer: ${version}`);
  return Object.freeze({ name, version, id: `${name}@${version}`, schema });
}
```

`src/kernel/predicates.ts`:

```ts
export interface Predicate<T = unknown> {
  /** Dotted lower-case name, e.g. `text.sanitised`. Guarantees and requirements match by id. */
  readonly id: string;
  readonly description: string;
  readonly test: (payload: T) => boolean;
}

export function definePredicate<T>(id: string, description: string, test: (payload: T) => boolean): Predicate<T> {
  if (!/^[a-z][a-z0-9-]*(\.[a-z][a-z0-9-]*)+$/.test(id)) {
    throw new TypeError(`predicate id must be dotted lower-case, e.g. text.sanitised: ${id}`);
  }
  return Object.freeze({ id, description, test });
}
```

```ts
import { z } from 'zod';
import { hashOf } from './canonical.js';
import { DATA_CLASSES, type DataClass } from './data-class.js';
import { ERROR_CODES, RETRYABLE, type ErrorCode } from './errors.js';
import type { Predicate } from './predicates.js';
import type { MessageType } from './types.js';

export const AGENT_KINDS = ['deterministic', 'llm', 'io', 'io-session', 'human', 'effect'] as const;
export type AgentKind = (typeof AGENT_KINDS)[number];

export interface Capabilities {
  readonly http?: true;
  readonly llm?: { readonly model: string; readonly maxTokens: number };
}

export interface ObligationSpec {
  readonly id: string;
  /** Agent id that must discharge it before the item may finish. */
  readonly dischargedBy: string;
}

export interface Contract<I = unknown, O = unknown> {
  readonly id: string;
  readonly version: string;
  readonly description: string;
  readonly skills: readonly string[];
  readonly team: string;
  readonly kind: AgentKind;
  readonly consumes: MessageType<I>;
  readonly produces: MessageType<O>;
  readonly requires: readonly Predicate<I>[];
  readonly guarantees: readonly Predicate<O>[];
  readonly capabilities: Capabilities;
  readonly egress: readonly string[];
  readonly dataClasses: { readonly in: readonly DataClass[]; readonly out: readonly DataClass[] };
  readonly redactor: boolean;
  readonly obligations: { readonly adds: readonly ObligationSpec[]; readonly discharges: readonly string[] };
  readonly budget: { readonly usdPerCall: number; readonly usdPerRun: number };
  readonly timeoutMs: number;
  readonly retry: { readonly max: number; readonly on: readonly ErrorCode[]; readonly backoffMs: readonly number[] };
  readonly breaker: { readonly consecutiveFailures: number };
  readonly guardrails: { readonly in: readonly string[]; readonly out: readonly string[] };
}

/** Predicates are contravariant in their payload, so heterogeneous collections use this. */
export type AnyContract = Contract<any, any>;

type Defaulted = 'capabilities' | 'egress' | 'redactor' | 'obligations' | 'retry' | 'breaker' | 'guardrails' | 'requires' | 'guarantees';
export type ContractSpec<I, O> = Omit<Contract<I, O>, Defaulted> & Partial<Pick<Contract<I, O>, Defaulted>>;

const kebab = /^[a-z][a-z0-9-]*$/;
const dotted = /^[a-z][a-z0-9.-]*$/;
const guardrailId = /^G-[A-Z0-9-]+$/;

const Meta = z.object({
  id: z.string().regex(kebab, 'id must be kebab-case'),
  version: z.string().regex(/^\d+\.\d+\.\d+$/, 'version must be semver x.y.z'),
  description: z
    .string()
    .min(10, 'description must say what the agent does')
    .refine((d) => !/\band\b/i.test(d), 'description contains "and": one agent does one skill, so split it'),
  skills: z.array(z.string().min(1)).min(1),
  team: z.string().regex(kebab, 'team must be kebab-case'),
  kind: z.enum(AGENT_KINDS),
  capabilities: z
    .object({
      http: z.literal(true).optional(),
      llm: z.object({ model: z.string().min(1), maxTokens: z.number().int().positive() }).strict().optional(),
    })
    .strict(),
  egress: z.array(z.string().regex(/^[a-z0-9.-]+$/, 'egress entries are bare host names')),
  dataClasses: z.object({ in: z.array(z.enum(DATA_CLASSES)).min(1), out: z.array(z.enum(DATA_CLASSES)).min(1) }),
  redactor: z.boolean(),
  obligations: z.object({
    adds: z.array(z.object({ id: z.string().regex(dotted), dischargedBy: z.string().regex(kebab) }).strict()),
    discharges: z.array(z.string().regex(dotted)),
  }),
  guardrails: z.object({ in: z.array(z.string().regex(guardrailId)), out: z.array(z.string().regex(guardrailId)) }),
  budget: z
    .object({ usdPerCall: z.number().nonnegative(), usdPerRun: z.number().nonnegative() })
    .refine((b) => b.usdPerRun >= b.usdPerCall, 'budget.usdPerRun must be at least budget.usdPerCall'),
  timeoutMs: z.number().int().positive().max(600_000),
  retry: z
    .object({
      max: z.number().int().min(0).max(5),
      on: z.array(z.enum(ERROR_CODES)),
      backoffMs: z.array(z.number().int().nonnegative()),
    })
    .refine((r) => r.backoffMs.length === r.max, 'retry.backoffMs needs exactly one entry per retry')
    .refine((r) => r.max === 0 || r.on.length > 0, 'retry.max > 0 needs at least one code in retry.on')
    .refine((r) => r.on.every((c) => RETRYABLE.has(c)), 'retry.on may only list retryable codes'),
  breaker: z.object({ consecutiveFailures: z.number().int().min(1) }),
});

export function defineContract<I, O>(spec: ContractSpec<I, O>): Contract<I, O> {
  const c: Contract<I, O> = {
    capabilities: {},
    egress: [],
    redactor: false,
    obligations: { adds: [], discharges: [] },
    retry: { max: 0, on: [], backoffMs: [] },
    breaker: { consecutiveFailures: 5 },
    guardrails: { in: [], out: [] },
    requires: [],
    guarantees: [],
    ...spec,
  };
  const parsed = Meta.safeParse(c);
  if (!parsed.success) {
    const msg = parsed.error.issues.map((i) => `${i.path.join('.') || '(root)'}: ${i.message}`).join('; ');
    throw new TypeError(`contract ${spec.id}: ${msg}`);
  }
  checkKindCapabilities(c);
  return deepFreeze(snapshot(c));
}

function deepFreeze<T>(v: T): T {
  if (v !== null && typeof v === 'object' && !Object.isFrozen(v)) {
    Object.freeze(v);
    for (const child of Object.values(v)) deepFreeze(child);
  }
  return v;
}

/** Copies every array and plain object the spec owns. consumes/produces/predicates are shared by reference. */
function snapshot<I, O>(c: Contract<I, O>): Contract<I, O> {
  const { capabilities } = c;
  return {
    ...c,
    skills: [...c.skills],
    requires: [...c.requires],
    guarantees: [...c.guarantees],
    capabilities: {
      ...(capabilities.http ? { http: capabilities.http } : {}),
      ...(capabilities.llm ? { llm: { ...capabilities.llm } } : {}),
    },
    egress: [...c.egress],
    dataClasses: { in: [...c.dataClasses.in], out: [...c.dataClasses.out] },
    obligations: { adds: c.obligations.adds.map((a) => ({ ...a })), discharges: [...c.obligations.discharges] },
    budget: { ...c.budget },
    retry: { max: c.retry.max, on: [...c.retry.on], backoffMs: [...c.retry.backoffMs] },
    breaker: { ...c.breaker },
    guardrails: { in: [...c.guardrails.in], out: [...c.guardrails.out] },
  };
}

function checkKindCapabilities(c: AnyContract): void {
  const problems: string[] = [];
  if (c.kind === 'llm' && !c.capabilities.llm) problems.push('kind llm requires capabilities.llm');
  if (c.kind !== 'llm' && c.capabilities.llm) problems.push(`kind ${c.kind} may not request capabilities.llm`);
  const httpKinds: readonly AgentKind[] = ['io', 'io-session', 'effect'];
  if (c.capabilities.http && !httpKinds.includes(c.kind)) problems.push(`kind ${c.kind} may not request capabilities.http`);
  if (c.capabilities.http && c.egress.length === 0) problems.push('capabilities.http requires at least one egress host');
  if (c.egress.length > 0 && !c.capabilities.http && !c.capabilities.llm) problems.push('egress requires capabilities.http or capabilities.llm');
  if ((c.kind === 'deterministic' || c.kind === 'human') && c.egress.length > 0) problems.push(`kind ${c.kind} may not declare egress`);
  if (c.kind === 'deterministic' && c.budget.usdPerCall > 0) problems.push('kind deterministic has no spend, so budget must be 0');
  if (problems.length > 0) throw new TypeError(`contract ${c.id}: ${problems.join('; ')}`);
}

export function contractMajor(c: Pick<Contract, 'version'>): string {
  return c.version.split('.')[0] ?? '0';
}

/** The interface (types, predicates, capabilities, data classes) and the governance settings (budget, timeout, retry, breaker, guardrails, obligations). Changing any of it without a version bump is drift. */
export function contractShape(c: AnyContract): Record<string, unknown> {
  return {
    id: c.id,
    version: c.version,
    kind: c.kind,
    team: c.team,
    consumes: { type: c.consumes.id, schema: z.toJSONSchema(c.consumes.schema) },
    produces: { type: c.produces.id, schema: z.toJSONSchema(c.produces.schema) },
    requires: c.requires.map((p) => p.id).sort(),
    guarantees: c.guarantees.map((p) => p.id).sort(),
    capabilities: c.capabilities,
    egress: [...c.egress].sort(),
    dataClasses: { in: [...c.dataClasses.in].sort(), out: [...c.dataClasses.out].sort() },
    redactor: c.redactor,
    obligations: {
      adds: [...c.obligations.adds].sort((a, b) => a.id.localeCompare(b.id)).map((a) => ({ id: a.id, dischargedBy: a.dischargedBy })),
      discharges: [...c.obligations.discharges].sort(),
    },
    budget: c.budget,
    timeoutMs: c.timeoutMs,
    breaker: c.breaker,
    retry: { max: c.retry.max, on: [...c.retry.on].sort(), backoffMs: c.retry.backoffMs },
    guardrails: { in: c.guardrails.in, out: c.guardrails.out },
  };
}

export function contractHash(c: AnyContract): string {
  return hashOf(contractShape(c));
}
```

```ts
import { sha256 } from './canonical.js';
import type { Contract } from './contract.js';
import type { DataClass } from './data-class.js';

export interface Obligation {
  readonly id: string;
  readonly dischargedBy: string;
  readonly discharged: boolean;
}

/** What one agent passes to the next, besides the payload. */
export interface Handoff {
  readonly guarantees: readonly string[];
  readonly dataClass: DataClass;
  readonly obligations: readonly Obligation[];
  readonly pins: { readonly policyHash: string };
}

export interface Producer {
  readonly agentId: string;
  readonly version: string;
  readonly contractHash: string;
}

export interface GuardrailVerdict {
  readonly id: string;
  readonly verdict: 'pass' | 'warn' | 'trip';
  readonly detail?: string;
}

export interface Cost {
  readonly usd: number;
  readonly inputTokens: number;
  readonly outputTokens: number;
}

export const ZERO_COST: Cost = Object.freeze({ usd: 0, inputTokens: 0, outputTokens: 0 });

/** Deliberately free of run ids and timestamps, so the same input yields byte-identical envelopes. */
export interface Envelope<T = unknown> {
  readonly envelopeId: string;
  readonly stepKey: string;
  readonly index: number;
  readonly type: string;
  readonly producer: Producer;
  readonly parents: readonly string[];
  readonly payload: T;
  readonly handoff: Handoff;
  readonly guardrails: readonly GuardrailVerdict[];
  readonly cost: Cost;
}

export function makeEnvelopeId(stepKey: string, index: number): string {
  return sha256(`${stepKey}:${index}`);
}

/** Discharge what this agent may discharge, add what it adds; sorted by id. */
export function nextObligations(input: readonly Obligation[], contract: Pick<Contract, 'id' | 'obligations'>): Obligation[] {
  const byId = new Map<string, Obligation>();
  for (const o of input) {
    const discharge = o.dischargedBy === contract.id && contract.obligations.discharges.includes(o.id);
    byId.set(o.id, discharge ? { ...o, discharged: true } : o);
  }
  for (const a of contract.obligations.adds) {
    if (!byId.has(a.id)) byId.set(a.id, { id: a.id, dischargedBy: a.dischargedBy, discharged: false });
  }
  return [...byId.values()].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
}

export function openObligations(env: Pick<Envelope, 'handoff'>): string[] {
  return env.handoff.obligations.filter((o) => !o.discharged).map((o) => o.id);
}
```

```ts
import { z } from 'zod';
import { hashOf } from './canonical.js';
import { DATA_CLASSES, type DataClass } from './data-class.js';
import { PolicyViolation } from './errors.js';

export interface Policy {
  readonly version: string;
  /** Hosts no agent may contact. */
  readonly deniedEgressHosts: readonly string[];
  /** Hosts no effect agent may submit to. */
  readonly deniedSubmitHosts: readonly string[];
  /** Data classes allowed to leave the machine, per host; hosts not listed use the default. */
  readonly egressClasses: Readonly<Record<string, readonly DataClass[]>>;
  readonly defaultEgressClasses: readonly DataClass[];
  readonly maxUsdPerRun: number;
  readonly maxPayloadBytes: number;
  readonly maxSubmitsPerDay: number;
  readonly requireApprovalForEffects: boolean;
}

/** Lower-case and strip trailing dots, so `Tracker.Example.` and `tracker.example` are one host. */
export function normalizeHost(host: string): string {
  return host.toLowerCase().replace(/\.+$/, '');
}

function deepFreeze<T>(value: T): T {
  if (typeof value === 'object' && value !== null && !Object.isFrozen(value)) {
    Object.freeze(value);
    for (const v of Object.values(value)) deepFreeze(v);
  }
  return value;
}

/** The product safety floor. A user policy may tighten it, never loosen it. */
export const PRODUCT_POLICY_V1: Policy = deepFreeze<Policy>({
  version: 'product.v1',
  deniedEgressHosts: [],
  deniedSubmitHosts: ['linkedin.com', 'naukri.com', 'indeed.com'],
  egressClasses: { 'api.anthropic.com': ['P0'] },
  defaultEgressClasses: ['P0'],
  maxUsdPerRun: 5,
  maxPayloadBytes: 262_144,
  maxSubmitsPerDay: 10,
  requireApprovalForEffects: true,
});

const HostName = z.string().regex(/^[A-Za-z0-9.-]+$/);

const UserPolicy = z
  .object({
    deniedEgressHosts: z.array(HostName).optional(),
    deniedSubmitHosts: z.array(HostName).optional(),
    egressClasses: z.record(HostName, z.array(z.enum(DATA_CLASSES))).optional(),
    maxUsdPerRun: z.number().nonnegative().optional(),
    maxSubmitsPerDay: z.number().int().nonnegative().optional(),
    requireApprovalForEffects: z.boolean().optional(),
  })
  .strict();
export type UserPolicy = z.infer<typeof UserPolicy>;

const union = (a: readonly string[], b: readonly string[] = []): string[] =>
  [...new Set([...a, ...b].map(normalizeHost))].sort();

function classesFor(p: Policy, host: string): readonly DataClass[] {
  return Object.hasOwn(p.egressClasses, host) ? (p.egressClasses[host] ?? p.defaultEgressClasses) : p.defaultEgressClasses;
}

/** effective = product ∧ user. Throws PolicyViolation listing every rule the user file tries to relax. */
export function mergePolicy(product: Policy, userRaw: unknown): Policy {
  const parsed = UserPolicy.safeParse(userRaw ?? {});
  const issues = parsed.success ? [] : parsed.error.issues.map((i) => `${i.path.map(String).join('.')}: ${i.message}`);
  // Zod's record silently drops a `__proto__` key, so check the raw keys ourselves.
  const rawClasses = (userRaw as { egressClasses?: unknown } | null | undefined)?.egressClasses;
  if (typeof rawClasses === 'object' && rawClasses !== null) {
    for (const k of Object.keys(rawClasses)) {
      if (!HostName.safeParse(k).success) issues.push(`egressClasses.${k}: invalid host name`);
    }
  }
  if (!parsed.success || issues.length > 0) throw new PolicyViolation(issues);
  const user = parsed.data;
  const violations: string[] = [];
  if (user.maxUsdPerRun !== undefined && user.maxUsdPerRun > product.maxUsdPerRun) {
    violations.push(`maxUsdPerRun may only be lowered (product ${product.maxUsdPerRun}, user ${user.maxUsdPerRun})`);
  }
  if (user.maxSubmitsPerDay !== undefined && user.maxSubmitsPerDay > product.maxSubmitsPerDay) {
    violations.push(`maxSubmitsPerDay may only be lowered (product ${product.maxSubmitsPerDay}, user ${user.maxSubmitsPerDay})`);
  }
  if (user.requireApprovalForEffects === false && product.requireApprovalForEffects) {
    violations.push('requireApprovalForEffects cannot be turned off');
  }
  const entries = new Map<string, readonly DataClass[]>(
    Object.entries(product.egressClasses).map(([h, c]) => [normalizeHost(h), [...c]]),
  );
  for (const [rawHost, classes] of Object.entries(user.egressClasses ?? {})) {
    const host = normalizeHost(rawHost);
    const allowed = classesFor(product, host);
    const extra = classes.filter((c) => !allowed.includes(c));
    if (extra.length > 0) violations.push(`egressClasses.${host} may not add ${extra.join(', ')}`);
    entries.set(host, [...new Set(classes)].sort());
  }
  if (violations.length > 0) throw new PolicyViolation(violations);
  return deepFreeze<Policy>({
    ...product,
    version: `${product.version}+user`,
    deniedEgressHosts: union(product.deniedEgressHosts, user.deniedEgressHosts),
    deniedSubmitHosts: union(product.deniedSubmitHosts, user.deniedSubmitHosts),
    egressClasses: Object.fromEntries(entries),
    defaultEgressClasses: [...product.defaultEgressClasses],
    maxUsdPerRun: user.maxUsdPerRun ?? product.maxUsdPerRun,
    maxSubmitsPerDay: user.maxSubmitsPerDay ?? product.maxSubmitsPerDay,
  });
}

export function policyHash(p: Policy): string {
  return hashOf(p);
}

export function hostMatches(host: string, pattern: string): boolean {
  const h = normalizeHost(host);
  const pat = normalizeHost(pattern);
  return h === pat || h.endsWith(`.${pat}`);
}

export function egressAllowed(p: Policy, host: string, cls: DataClass): boolean {
  const h = normalizeHost(host);
  if (p.deniedEgressHosts.some((d) => hostMatches(h, d))) return false;
  return classesFor(p, h).includes(cls);
}
```

```ts
import type { AnyContract } from './contract.js';
import type { DataClass } from './data-class.js';
import type { Cost } from './envelope.js';
import { KernelError } from './errors.js';

  getText(url: string): Promise<{ status: number; body: string }>;

export interface LlmTool {
  complete(req: { system: string; prompt: string }): Promise<LlmResponse>;

/**
 * Everything an agent may touch. Tools exist only when the contract grants
 * them. There is deliberately no proxy, user-agent, fingerprint or CAPTCHA
 * capability: those are absent, not disabled (G-NO-EVASION).
 */
export interface AgentContext {
  readonly clock: Clock;
  readonly signal: AbortSignal;
  readonly http?: HttpTool;
  readonly llm?: LlmTool;
  spent(): Cost;
}

export interface IssueArgs {
  readonly contract: AnyContract;
  readonly policy: Policy;
  readonly inputClass: DataClass;
  readonly providers: Providers;
  readonly clock: Clock;
  readonly signal: AbortSignal;
}

function mapProviderError(e: unknown): unknown {
  if (e instanceof KernelError) return e;
  const o = typeof e === 'object' && e !== null ? (e as { name?: unknown; status?: unknown; message?: unknown }) : {};
  if (o.name === 'TimeoutError') return e;
  if (typeof o.status === 'number') {
    if (o.status === 429) return new KernelError('rate_limit', `provider returned ${o.status}`);
    if (o.status >= 500) return new KernelError('overloaded', `provider returned ${o.status}`);
  }
  if (o.name === 'AbortError') return new KernelError('unknown', 'request aborted');
  return new KernelError('network', typeof o.message === 'string' ? o.message : 'provider call failed');
}

export function issueContext(a: IssueArgs): AgentContext {
  let usd = 0;
  let inputTokens = 0;
  let outputTokens = 0;

  const checkHost = (host: string): void => {
    if (!a.contract.egress.includes(host)) {
      throw new KernelError('policy_denied', `G-EGRESS: ${a.contract.id} may not contact ${host}`);
    }
    if (!egressAllowed(a.policy, host, a.inputClass)) {
      throw new KernelError('policy_denied', `G-PII-EGRESS: ${a.inputClass} data may not be sent to ${host}`);
    }
  };

  const ctx: { -readonly [K in keyof AgentContext]: AgentContext[K] } = {
    clock: a.clock,
    signal: a.signal,
    spent: () => ({ usd, inputTokens, outputTokens }),
  };

  if (a.contract.capabilities.http) {
    ctx.http = {
      async getText(url) {
        let u: URL;
        try {
          u = new URL(url);
        } catch {
          throw new KernelError('policy_denied', 'G-EGRESS: invalid URL');
        }
        if (u.protocol !== 'https:') throw new KernelError('policy_denied', `G-EGRESS: only https is allowed (${url})`);
        if (u.port !== '') throw new KernelError('policy_denied', `G-EGRESS: explicit ports are not allowed (${u.hostname})`);
        if (u.username !== '' || u.password !== '') throw new KernelError('policy_denied', 'G-EGRESS: credentials in URLs are not allowed');
        checkHost(u.hostname);
        if (a.signal.aborted) throw new KernelError('timeout', 'step timed out; no new requests');
        let res;
        let body;
        try {
          res = await a.providers.fetch(u.href, { signal: a.signal, redirect: 'manual' });
          if (res.status >= 300 && res.status < 400) {
            throw new KernelError('policy_denied', `G-EGRESS: redirect from ${u.hostname} refused`);
          }
          if (res.status === 429) throw new KernelError('rate_limit', `429 from ${u.hostname}`);
          if (res.status >= 500) throw new KernelError('overloaded', `${res.status} from ${u.hostname}`);
          body = await res.text();
        } catch (e) {
          throw mapProviderError(e);
        }
        return { status: res.status, body };
      },
    };
  }

  const llmCap = a.contract.capabilities.llm;
  const provider = a.providers.llm;
  if (llmCap) {
    let overCall = false;
    let inFlight = false;
    ctx.llm = {
      async complete(req) {
        if (!provider) throw new KernelError('policy_denied', 'no LLM provider is configured');
        checkHost(provider.host);
        if (inFlight) throw new KernelError('policy_denied', 'G-BUDGET: one model call at a time per step');
        if (a.signal.aborted) throw new KernelError('timeout', 'step timed out; no new model calls');
        inFlight = true;
        try {
          if (overCall) throw new KernelError('budget_exceeded', 'G-BUDGET: a previous call exceeded usdPerCall');
          if (usd >= a.contract.budget.usdPerRun) {
            throw new KernelError('budget_exceeded', 'G-BUDGET: usdPerRun reached in this step');
          }
          let res;
          try {
            res = await provider.complete({ ...req, model: llmCap.model, maxTokens: llmCap.maxTokens }, a.signal);
          } catch (e) {
            // A failed provider call reports no usage, so spend is left unchanged.
            throw mapProviderError(e);
          }
          usd += res.usd;
          inputTokens += res.inputTokens;
          outputTokens += res.outputTokens;
          if (res.usd > a.contract.budget.usdPerCall) {
            overCall = true;
            throw new KernelError('budget_exceeded', `G-BUDGET: call cost ${res.usd} > usdPerCall ${a.contract.budget.usdPerCall}`);
          }
          return res;
        } finally {
          inFlight = false;
        }
      },
    };
  }
  return ctx;
}
```

```ts
import type { AnyContract } from '../contract.js';
import type { DataClass } from '../data-class.js';
import type { Envelope, GuardrailVerdict } from '../envelope.js';
import type { Policy } from '../policy.js';

export type Side = 'in' | 'out';

export interface GuardrailContext {
  readonly side: Side;
  readonly contract: AnyContract;
  readonly policy: Policy;
  readonly policyHash: string;
  /** Resolves the registered contract hash for a producer; undefined if unknown. */
  readonly contractHashOf: (agentId: string, version: string) => string | undefined;
  readonly input: Envelope;
  /** Present on the out side: one output payload and its computed data class. */
  readonly output?: { readonly payload: unknown; readonly dataClass: DataClass };
}

/** A pure function of the envelope, the contract and the policy. */
export interface Guardrail {
  readonly id: string;
  readonly sides: readonly Side[];
  check(ctx: GuardrailContext): GuardrailVerdict;
}

/** These run on every step whether or not a contract lists them: the kernel fails closed. */
export const MANDATORY: Readonly<Record<Side, readonly string[]>> = {
  in: ['G-SCHEMA', 'G-LINEAGE', 'G-DATACLASS', 'G-SIZE', 'G-PRE'],
  out: ['G-SCHEMA', 'G-DATACLASS', 'G-POST'],
};

export class GuardrailRegistry {
  private readonly byId = new Map<string, Guardrail>();

  constructor(guardrails: readonly Guardrail[] = []) {
    for (const g of guardrails) this.add(g);
  }

  add(g: Guardrail): void {
    if (this.byId.has(g.id)) throw new Error(`guardrail ${g.id} registered twice`);
    this.byId.set(g.id, g);
  }

  has(id: string): boolean {
    return this.byId.has(id);
  }

  /** Mandatory ids first, then the contract's own, deduplicated; stops at the first trip. */
  run(side: Side, ctx: GuardrailContext): { verdicts: GuardrailVerdict[]; tripped: boolean } {
    const declared = side === 'in' ? ctx.contract.guardrails.in : ctx.contract.guardrails.out;
    const ids = [...new Set([...MANDATORY[side], ...declared])];
    const verdicts: GuardrailVerdict[] = [];
    for (const id of ids) {
      const g = this.byId.get(id);
      if (!g) {
        verdicts.push({ id, verdict: 'trip', detail: 'unknown guardrail' });
        return { verdicts, tripped: true };
      }
      if (!g.sides.includes(side)) {
        if (MANDATORY[side].includes(id)) {
          verdicts.push({ id, verdict: 'trip', detail: `mandatory guardrail does not run on the ${side} side` });
          return { verdicts, tripped: true };
        }
        continue;
      }
      let v: GuardrailVerdict;
      try {
        v = g.check({ ...ctx, side });
      } catch (e) {
        v = { id, verdict: 'trip', detail: `guardrail threw: ${e instanceof Error ? e.message : String(e)}` };
      }
      const kind: unknown = (v as { verdict?: unknown } | null | undefined)?.verdict;
      const validId = (v as { id?: unknown } | null | undefined)?.id === id;
      if (!validId || (kind !== 'pass' && kind !== 'warn' && kind !== 'trip')) {
        verdicts.push({ id, verdict: 'trip', detail: 'guardrail returned an invalid verdict' });
        return { verdicts, tripped: true };
      }
      verdicts.push(v);
      if (v.verdict === 'trip') return { verdicts, tripped: true };
    }
    return { verdicts, tripped: false };
  }
}
```

```ts
import { canonicalJson } from '../canonical.js';
import type { GuardrailVerdict } from '../envelope.js';
import { type Guardrail, GuardrailRegistry } from './index.js';

const pass = (id: string): GuardrailVerdict => ({ id, verdict: 'pass' });
const trip = (id: string, detail: string): GuardrailVerdict => ({ id, verdict: 'trip', detail });

/** Strict parse, and the parse must not change the payload: no coercion, no defaults, no repair. */
export const G_SCHEMA: Guardrail = {
  id: 'G-SCHEMA',
  sides: ['in', 'out'],
  check(ctx) {
    if (ctx.side === 'out' && !ctx.output) return trip('G-SCHEMA', 'no output to check');
    const type = ctx.side === 'in' ? ctx.contract.consumes : ctx.contract.produces;
    const payload = ctx.side === 'in' ? ctx.input.payload : ctx.output?.payload;
    if (ctx.side === 'in' && ctx.input.type !== type.id) return trip('G-SCHEMA', `envelope type ${ctx.input.type}, contract consumes ${type.id}`);
    const r = type.schema.safeParse(payload);
    if (!r.success) {
      const i = r.error.issues[0];
      return trip('G-SCHEMA', `${type.id}: ${i ? `${i.path.join('.') || '(root)'} ${i.message}` : 'invalid'}`);
    }
    if (canonicalJson(r.data) !== canonicalJson(payload)) return trip('G-SCHEMA', `${type.id}: schema would coerce the payload`);
    return pass('G-SCHEMA');
  },
};

export const G_LINEAGE: Guardrail = {
  id: 'G-LINEAGE',
  sides: ['in'],
  check(ctx) {
    const p = ctx.input.producer;
    const known = ctx.contractHashOf(p.agentId, p.version);
    if (known === undefined) return trip('G-LINEAGE', `unknown producer ${p.agentId}@${p.version}`);
    if (known !== p.contractHash) return trip('G-LINEAGE', `${p.agentId}@${p.version} contract hash does not match the registry`);
    if (ctx.input.handoff.pins.policyHash !== ctx.policyHash) return trip('G-LINEAGE', 'envelope was produced under a different policy');
    return pass('G-LINEAGE');
  },
};

export const G_DATACLASS: Guardrail = {
  id: 'G-DATACLASS',
  sides: ['in', 'out'],
  check(ctx) {
    if (ctx.side === 'in') {
      const c = ctx.input.handoff.dataClass;
      return ctx.contract.dataClasses.in.includes(c) ? pass('G-DATACLASS') : trip('G-DATACLASS', `${ctx.contract.id} may not receive ${c} data`);
    }
    if (!ctx.output) return trip('G-DATACLASS', 'no output to check');
    const c = ctx.output.dataClass;
    return ctx.contract.dataClasses.out.includes(c) ? pass('G-DATACLASS') : trip('G-DATACLASS', `output would carry ${c}, contract emits ${ctx.contract.dataClasses.out.join('/')}`);
  },
};

export const G_SIZE: Guardrail = {
  id: 'G-SIZE',
  sides: ['in'],
  check(ctx) {
    const bytes = Buffer.byteLength(canonicalJson(ctx.input.payload), 'utf8');
    return bytes <= ctx.policy.maxPayloadBytes ? pass('G-SIZE') : trip('G-SIZE', `${bytes} bytes > ${ctx.policy.maxPayloadBytes}`);
  },
};

/** The producer must have promised it, and it must hold on the actual payload. */
export const G_PRE: Guardrail = {
  id: 'G-PRE',
  sides: ['in'],
  check(ctx) {
    for (const p of ctx.contract.requires) {
      if (!ctx.input.handoff.guarantees.includes(p.id)) return trip('G-PRE', `${p.id} was not guaranteed by ${ctx.input.producer.agentId}`);
      if (!p.test(ctx.input.payload)) return trip('G-PRE', `${p.id} does not hold on the input`);
    }
    return pass('G-PRE');
  },
};

export const G_POST: Guardrail = {
  id: 'G-POST',
  sides: ['out'],
  check(ctx) {
    if (!ctx.output) return trip('G-POST', 'no output to check');
    for (const p of ctx.contract.guarantees) {
      if (!p.test(ctx.output.payload)) return trip('G-POST', `${p.id} does not hold on the output`);
    }
    return pass('G-POST');
  },
};

export const BUILTIN_GUARDRAILS: readonly Guardrail[] = [G_SCHEMA, G_LINEAGE, G_DATACLASS, G_SIZE, G_PRE, G_POST];

export function builtinGuardrails(): GuardrailRegistry {
  return new GuardrailRegistry(BUILTIN_GUARDRAILS);
}
```

```ts
import type { AnyAgent } from './agent.js';
import { contractHash } from './contract.js';

export class Registry {
  private readonly agents = new Map<string, AnyAgent>();
  private readonly hashes = new Map<string, string>();

  register(agent: AnyAgent): void {
    const c = agent.contract;
    if (this.agents.has(c.id)) throw new Error(`agent ${c.id} registered twice`);
    this.agents.set(c.id, agent);
    this.hashes.set(`${c.id}@${c.version}`, contractHash(c));
  }

  get(id: string): AnyAgent {
    const a = this.agents.get(id);
    if (!a) throw new Error(`unknown agent ${id}`);
    return a;
  }

  has(id: string): boolean {
    return this.agents.has(id);
  }

  /** Sorted by id so iteration order never depends on registration order. */
  list(): AnyAgent[] {
    return [...this.agents.values()].sort((a, b) => (a.contract.id < b.contract.id ? -1 : 1));
  }

  contractHashOf(agentId: string, version: string): string | undefined {
    return this.hashes.get(`${agentId}@${version}`);
  }
}
```

```ts
import { z } from 'zod';
import { hashOf } from './canonical.js';
import { WiringError } from './errors.js';
import type { GuardrailRegistry } from './guardrails/index.js';
import type { Predicate } from './predicates.js';
import type { Registry } from './registry.js';
import type { MessageType } from './types.js';

export const SUPERVISOR_ID = 'supervisor';

export interface Team {
  readonly id: string;
  /** Name of the team lead; its playbook is the team's edges. */
  readonly lead: string;
  readonly members: readonly string[];
  /** Every envelope leaving the team must carry these guarantees. */
  readonly exitGuarantees: readonly string[];
  readonly breaker: { readonly consecutiveFailures: number };
}

export interface Edge {
  /** Message type id, e.g. `DemoItem@1`. */
  readonly type: string;
  readonly to: string;
}

export interface Pipeline {
  readonly id: string;
  readonly version: string;
  /** Seed payload type and the guarantees the supervisor checks on it (any: predicates are contravariant). */
  readonly seed: { readonly type: MessageType<any>; readonly guarantees: readonly Predicate<any>[] };
  readonly teams: readonly Team[];
  readonly edges: readonly Edge[];
}

export interface CompiledPipeline {
  readonly pipeline: Pipeline;
  /** type id → consumer agent ids, sorted. */
  readonly routes: ReadonlyMap<string, readonly string[]>;
  /** agent id → depth from the seed; queues drain in stage order. */
  readonly stage: ReadonlyMap<string, number>;
  readonly teamOf: ReadonlyMap<string, Team>;
  /** Stands in for a contract hash on seed envelopes. */
  readonly seedHash: string;
}

/** Proves the graph is sound before anything runs. Collects every problem, then throws once. */
export function compileWiring(registry: Registry, pipeline: Pipeline, guardrails: GuardrailRegistry): CompiledPipeline {
  const problems: string[] = [];
  const teamOf = new Map<string, Team>();
  const teamIds = new Set<string>();

  for (const t of pipeline.teams) {
    if (teamIds.has(t.id)) problems.push(`team ${t.id} declared twice`);
    teamIds.add(t.id);
    for (const m of t.members) {
      if (teamOf.has(m)) problems.push(`agent ${m} is in two teams`);
      teamOf.set(m, t);
      if (!registry.has(m)) {
        problems.push(`team ${t.id} lists unregistered agent ${m}`);
        continue;
      }
      const c = registry.get(m).contract;
      if (c.team !== t.id) problems.push(`agent ${m} declares team ${c.team} but is listed in ${t.id}`);
      for (const g of [...c.guardrails.in, ...c.guardrails.out]) {
        if (!guardrails.has(g)) problems.push(`agent ${m} lists unknown guardrail ${g}`);
      }
    }
  }

  const members = [...teamOf.keys()].filter((m) => registry.has(m));
  const producersOf = (type: string): string[] => {
    const ps = members.filter((m) => registry.get(m).contract.produces.id === type);
    if (pipeline.seed.type.id === type) ps.push(SUPERVISOR_ID);
    return ps.sort();
  };
  const guaranteesOf = (producer: string): string[] =>
    producer === SUPERVISOR_ID ? pipeline.seed.guarantees.map((g) => g.id) : registry.get(producer).contract.guarantees.map((g) => g.id);

  const routes = new Map<string, string[]>();
  for (const e of pipeline.edges) {
    if (!registry.has(e.to)) {
      problems.push(`edge ${e.type} → ${e.to}: agent is not registered`);
      continue;
    }
    if (!teamOf.has(e.to)) problems.push(`edge ${e.type} → ${e.to}: agent is in no team`);
    const c = registry.get(e.to).contract;
    if (c.consumes.id !== e.type) problems.push(`edge ${e.type} → ${e.to}: agent consumes ${c.consumes.id}`);
    const producers = producersOf(e.type);
    if (producers.length === 0) problems.push(`edge ${e.type} → ${e.to}: nothing produces ${e.type}`);
    for (const p of producers) {
      const promised = guaranteesOf(p);
      for (const r of c.requires) {
        if (!promised.includes(r.id)) problems.push(`${e.to} requires ${r.id} but producer ${p} does not guarantee it`);
      }
      const from = teamOf.get(p);
      const to = teamOf.get(e.to);
      if (p !== SUPERVISOR_ID && from && to && from.id !== to.id) {
        for (const g of from.exitGuarantees) {
          if (!promised.includes(g)) problems.push(`${p} leaves team ${from.id} without its exit guarantee ${g}`);
        }
      }
    }
    const list = routes.get(e.type) ?? [];
    if (list.includes(e.to)) problems.push(`edge ${e.type} → ${e.to} declared twice`);
    list.push(e.to);
    routes.set(e.type, list.sort());
  }

  checkSchemaIdentity(registry, pipeline, members, problems);
  if (!pipeline.edges.some((e) => e.type === pipeline.seed.type.id)) problems.push(`nothing consumes the seed type ${pipeline.seed.type.id}`);

  const stage = computeStages(registry, pipeline, routes, problems);
  if (problems.length > 0) throw new WiringError(problems);
  return {
    pipeline,
    routes,
    stage,
    teamOf,
    seedHash: hashOf({ pipeline: pipeline.id, version: pipeline.version, seed: pipeline.seed.type.id, guarantees: pipeline.seed.guarantees.map((g) => g.id).sort() }),
  };
}

/** Longest path from the seed; a cycle is a wiring error. */
function computeStages(registry: Registry, pipeline: Pipeline, routes: Map<string, string[]>, problems: string[]): Map<string, number> {
  const stage = new Map<string, number>();
  const visiting = new Set<string>();
  const visit = (agent: string, depth: number, path: string[]): void => {
    if (visiting.has(agent)) {
      problems.push(`cycle: ${[...path, agent].join(' → ')}`);
      return;
    }
    if ((stage.get(agent) ?? -1) >= depth) return;
    stage.set(agent, depth);
    if (!registry.has(agent)) return;
    visiting.add(agent);
    for (const next of routes.get(registry.get(agent).contract.produces.id) ?? []) visit(next, depth + 1, [...path, agent]);
    visiting.delete(agent);
  };
  for (const first of routes.get(pipeline.seed.type.id) ?? []) visit(first, 1, []);
  for (const t of pipeline.teams) {
    for (const m of t.members) if (registry.has(m) && !stage.has(m)) problems.push(`agent ${m} is unreachable from the seed`);
  }
  return stage;
}

/** One type id must mean one payload shape everywhere it is used. */
function checkSchemaIdentity(registry: Registry, pipeline: Pipeline, members: string[], problems: string[]): void {
  const uses = new Map<string, Map<string, string>>(); // type id → agent → schema hash
  const note = (type: MessageType<any>, agent: string): void => {
    const byAgent = uses.get(type.id) ?? new Map<string, string>();
    if (!byAgent.has(agent)) byAgent.set(agent, hashOf(z.toJSONSchema(type.schema)));
    uses.set(type.id, byAgent);
  };
  note(pipeline.seed.type, SUPERVISOR_ID);
  for (const m of members) {
    const c = registry.get(m).contract;
    note(c.consumes, m);
    note(c.produces, m);
  }
  for (const [id, byAgent] of [...uses].sort(([a], [b]) => (a < b ? -1 : 1))) {
    const agents = [...byAgent.keys()].sort();
    const first = agents[0];
    const other = agents.find((a) => byAgent.get(a) !== byAgent.get(first ?? ''));
    if (first !== undefined && other !== undefined) problems.push(`type ${id} is defined with different schemas (${first}, ${other})`);
  }
}
```

```ts
import { InvalidTransition } from './errors.js';

export interface MachineDef<S extends string, E extends string> {
  readonly id: string;
  /** State an entity is in before its first event. */
  readonly initial: S;
  readonly terminal: readonly S[];
  /** `from: '*'` matches any non-terminal state; exact matches win. */
  readonly transitions: readonly { readonly from: S | '*'; readonly on: E; readonly to: S }[];
}

export class StateMachine<S extends string = string, E extends string = string> {
  constructor(readonly def: MachineDef<S, E>) {
    const seen = new Set<string>();
    for (const t of def.transitions) {
      const k = `${t.from}|${t.on}`;
      if (seen.has(k)) throw new Error(`${def.id}: duplicate transition ${t.from} --${t.on}-->`);
      seen.add(k);
      if (t.from !== '*' && def.terminal.includes(t.from)) throw new Error(`${def.id}: terminal state ${t.from} has an outgoing transition`);
    }
  }

  next(from: S, on: E): S {
    const exact = this.def.transitions.find((t) => t.from === from && t.on === on);
    if (exact) return exact.to;
    if (!this.def.terminal.includes(from)) {
      const any = this.def.transitions.find((t) => t.from === '*' && t.on === on);
      if (any) return any.to;
    }
    throw new InvalidTransition(this.def.id, from, on);
  }

  isTerminal(s: S): boolean {
    return this.def.terminal.includes(s);
  }
}
```

```ts
/** Consecutive-failure breakers, per run. Keys are `agent:<id>` or `team:<id>`.
 * An open breaker stays open for the rest of the run; `success` resets the failure count but never closes an open breaker. */
export class Breakers {
  private readonly fails = new Map<string, number>();
  private readonly open = new Set<string>();

  isOpen(key: string): boolean {
    return this.open.has(key);
  }

  success(key: string): void {
    this.fails.set(key, 0);
  }

  /** Returns true when this failure opens the breaker. */
  failure(key: string, threshold: number): boolean {
    const n = (this.fails.get(key) ?? 0) + 1;
    this.fails.set(key, n);
    if (n >= threshold && !this.open.has(key)) {
      this.open.add(key);
      return true;
    }
    return false;
  }
}
```

```ts
import Database from 'better-sqlite3';
import { canonicalJson, sha256 } from '../canonical.js';
import type { Cost, Envelope } from '../envelope.js';
import type { ErrorCode } from '../errors.js';
import type { StateMachine } from '../state-machine.js';
import { MIGRATIONS } from './migrations.js';

export type StepStatus = 'running' | 'done' | 'failed' | 'quarantined';
export type QueueStatus = 'pending' | 'done' | 'failed' | 'quarantined' | 'skipped';

export interface StepRow {
  readonly step_key: string;
  readonly agent_id: string;
  readonly contract_version: string;
  readonly contract_hash: string;
  readonly input_envelope_id: string;
  readonly status: StepStatus;
  readonly attempts: number;
  readonly error_code: ErrorCode | null;
  readonly detail: string | null;
}

export interface QueueRow {
  readonly run_id: string;
  readonly stage: number;
  readonly agent_id: string;
  readonly envelope_id: string;
  readonly status: QueueStatus;
}

export interface EventRow {
  readonly seq: number;
  readonly run_id: string;
  readonly at: string;
  readonly kind: string;
  readonly data: string;
  readonly policy_hash: string;
  readonly prev_hash: string;
  readonly hash: string;
}

export const GENESIS_HASH = '0'.repeat(64);

export function eventHash(prevHash: string, e: Pick<EventRow, 'run_id' | 'at' | 'kind' | 'data' | 'policy_hash'>): string {
  return sha256(prevHash + canonicalJson({ run_id: e.run_id, at: e.at, kind: e.kind, data: e.data, policy_hash: e.policy_hash }));
}

/**
 * The only module that touches SQLite. better-sqlite3 is synchronous, so
 * writes happen in program order and every multi-row write is one transaction.
 */
export class Store {
  private constructor(readonly db: Database.Database) {}

  static open(path: string): Store {
    const db = new Database(path);
    db.pragma('journal_mode = WAL');
    db.pragma('foreign_keys = ON');
    try {
      const version = db.pragma('user_version', { simple: true }) as number;
      if (version > MIGRATIONS.length) {
        throw new Error(`database schema version ${version} is newer than this jobpilot (${MIGRATIONS.length})`);
      }
      for (let v = version; v < MIGRATIONS.length; v++) {
        db.transaction(() => {
          db.exec(MIGRATIONS[v] ?? '');
          db.pragma(`user_version = ${v + 1}`);
        })();
      }
    } catch (err) {
      db.close();
      throw err;
    }
    return new Store(db);
  }

  close(): void {
    this.db.close();
  }

  tx<T>(fn: () => T): T {
    return this.db.transaction(fn)();
  }

  // runs
  startRun(runId: string, pipeline: string, policyHash: string, at: string): void {
    this.db
      .prepare(`INSERT INTO runs (run_id, pipeline, policy_hash, started_at, status) VALUES (?, ?, ?, ?, 'running')`)
      .run(runId, pipeline, policyHash, at);
  }

  finishRun(runId: string, at: string): void {
    this.db.prepare(`UPDATE runs SET status = 'completed', finished_at = ? WHERE run_id = ?`).run(at, runId);
  }

  getRun(runId: string): { run_id: string; pipeline: string; policy_hash: string; status: string } | undefined {
    return this.db.prepare(`SELECT run_id, pipeline, policy_hash, status FROM runs WHERE run_id = ?`).get(runId) as
      | { run_id: string; pipeline: string; policy_hash: string; status: string }
      | undefined;
  }

  // steps
  getStep(stepKey: string): StepRow | undefined {
    return this.db.prepare(`SELECT * FROM steps WHERE step_key = ?`).get(stepKey) as StepRow | undefined;
  }

  markRunning(s: { stepKey: string; agentId: string; version: string; contractHash: string; inputEnvelopeId: string }): void {
    this.db
      .prepare(
        `INSERT INTO steps (step_key, agent_id, contract_version, contract_hash, input_envelope_id, status, attempts)
         VALUES (?, ?, ?, ?, ?, 'running', 1)
         ON CONFLICT (step_key) DO UPDATE SET agent_id = excluded.agent_id, contract_version = excluded.contract_version,
           contract_hash = excluded.contract_hash, input_envelope_id = excluded.input_envelope_id,
           status = 'running', attempts = attempts + 1, error_code = NULL, detail = NULL`,
      )
      .run(s.stepKey, s.agentId, s.version, s.contractHash, s.inputEnvelopeId);
  }

  /** Writes the step outcome, its envelopes, its cost and (when given) its hash-chained event atomically. */
  finishStep(a: {
    runId: string;
    stepKey: string;
    agentId: string;
    status: Exclude<StepStatus, 'running'>;
    envelopes: readonly Envelope[];
    cost: Cost;
    errorCode?: ErrorCode;
    detail?: string;
    event?: { at: string; kind: string; data: unknown; policyHash: string };
  }): void {
    this.tx(() => {
      const res = this.db
        .prepare(`UPDATE steps SET status = ?, error_code = ?, detail = ? WHERE step_key = ? AND status = 'running'`)
        .run(a.status, a.errorCode ?? null, a.detail ?? null, a.stepKey);
      if (res.changes !== 1) throw new Error(`finishStep: step ${a.stepKey} is not running`);
      const ins = this.db.prepare(`INSERT INTO envelopes (envelope_id, step_key, idx, type, body) VALUES (?, ?, ?, ?, ?)`);
      for (const e of a.envelopes) ins.run(e.envelopeId, e.stepKey, e.index, e.type, canonicalJson(e));
      if (a.cost.usd > 0 || a.cost.inputTokens > 0 || a.cost.outputTokens > 0) {
        this.db
          .prepare(`INSERT INTO cost_ledger (run_id, step_key, agent_id, usd, input_tokens, output_tokens) VALUES (?, ?, ?, ?, ?, ?)`)
          .run(a.runId, a.stepKey, a.agentId, a.cost.usd, a.cost.inputTokens, a.cost.outputTokens);
      }
      if (a.event) this.appendEvent(a.runId, a.event.at, a.event.kind, a.event.data, a.event.policyHash);
    });
  }

  /** For seed envelopes, which have no agent behind them. */
  putSeed(stepKey: string, env: Envelope, pipelineVersion: string, seedHash: string): void {
    this.tx(() => {
      this.db
        .prepare(
          `INSERT OR IGNORE INTO steps (step_key, agent_id, contract_version, contract_hash, input_envelope_id, status, attempts)
           VALUES (?, 'supervisor', ?, ?, '', 'done', 0)`,
        )
        .run(stepKey, pipelineVersion, seedHash);
      this.db
        .prepare(`INSERT OR IGNORE INTO envelopes (envelope_id, step_key, idx, type, body) VALUES (?, ?, ?, ?, ?)`)
        .run(env.envelopeId, env.stepKey, env.index, env.type, canonicalJson(env));
    });
  }

  envelopesOfStep(stepKey: string): Envelope[] {
    const rows = this.db.prepare(`SELECT body FROM envelopes WHERE step_key = ? ORDER BY idx`).all(stepKey) as { body: string }[];
    return rows.map((r) => JSON.parse(r.body) as Envelope);
  }

  getEnvelope(envelopeId: string): Envelope {
    const row = this.db.prepare(`SELECT body FROM envelopes WHERE envelope_id = ?`).get(envelopeId) as { body: string } | undefined;
    if (!row) throw new Error(`unknown envelope ${envelopeId}`);
    return JSON.parse(row.body) as Envelope;
  }

  // cost
  runCost(runId: string, agentId?: string): number {
    const row = (
      agentId === undefined
        ? this.db.prepare(`SELECT COALESCE(SUM(usd), 0) AS usd FROM cost_ledger WHERE run_id = ?`).get(runId)
        : this.db.prepare(`SELECT COALESCE(SUM(usd), 0) AS usd FROM cost_ledger WHERE run_id = ? AND agent_id = ?`).get(runId, agentId)
    ) as { usd: number };
    return row.usd;
  }

  // queue
  enqueue(runId: string, stage: number, agentId: string, envelopeId: string): void {
    this.db
      .prepare(`INSERT OR IGNORE INTO queue (run_id, stage, agent_id, envelope_id, status) VALUES (?, ?, ?, ?, 'pending')`)
      .run(runId, stage, agentId, envelopeId);
  }

  /** Total order: stage, then agent, then envelope id. Network timing never decides what runs next. */
  nextQueued(runId: string): QueueRow | undefined {
    return this.db
      .prepare(`SELECT * FROM queue WHERE run_id = ? AND status = 'pending' ORDER BY stage, agent_id, envelope_id LIMIT 1`)
      .get(runId) as QueueRow | undefined;
  }

  markQueue(runId: string, agentId: string, envelopeId: string, status: QueueStatus): void {
    this.db.prepare(`UPDATE queue SET status = ? WHERE run_id = ? AND agent_id = ? AND envelope_id = ?`).run(status, runId, agentId, envelopeId);
  }

  queueCounts(runId: string): Record<QueueStatus, number> {
    const rows = this.db.prepare(`SELECT status, COUNT(*) AS n FROM queue WHERE run_id = ? GROUP BY status`).all(runId) as { status: QueueStatus; n: number }[];
    const out: Record<QueueStatus, number> = { pending: 0, done: 0, failed: 0, quarantined: 0, skipped: 0 };
    for (const r of rows) out[r.status] = r.n;
    return out;
  }

  // events: append-only, hash-chained
  appendEvent(runId: string, at: string, kind: string, data: unknown, policyHash: string): void {
    this.tx(() => {
      const head = this.chainHead();
      const prev = head?.hash ?? GENESIS_HASH;
      const row = { run_id: runId, at, kind, data: canonicalJson(data), policy_hash: policyHash };
      const hash = eventHash(prev, row);
      const res = this.db
        .prepare(`INSERT INTO events (run_id, at, kind, data, policy_hash, prev_hash, hash) VALUES (?, ?, ?, ?, ?, ?, ?)`)
        .run(row.run_id, row.at, row.kind, row.data, row.policy_hash, prev, hash);
      this.db
        .prepare(
          `INSERT INTO chain_head (id, seq, hash, count) VALUES (1, ?, ?, 1)
           ON CONFLICT (id) DO UPDATE SET seq = excluded.seq, hash = excluded.hash, count = count + 1`,
        )
        .run(Number(res.lastInsertRowid), hash);
    });
  }

  /** The stored anchor of the audit chain: last seq, last hash and number of events appended. */
  chainHead(): { seq: number; hash: string; count: number } | undefined {
    return this.db.prepare(`SELECT seq, hash, count FROM chain_head WHERE id = 1`).get() as
      | { seq: number; hash: string; count: number }
      | undefined;
  }

  events(runId?: string): EventRow[] {
    return (
      runId === undefined
        ? this.db.prepare(`SELECT * FROM events ORDER BY seq`).all()
        : this.db.prepare(`SELECT * FROM events WHERE run_id = ? ORDER BY seq`).all(runId)
    ) as EventRow[];
  }

  // entities: idempotent per (entity, step, event), so a replayed step never double-applies
  applyTransition(entityId: string, kind: string, stepKey: string, event: string, machine: StateMachine): string | undefined {
    return this.tx(() => {
      const done = this.db
        .prepare(`SELECT 1 FROM entity_transitions WHERE entity_id = ? AND step_key = ? AND event = ?`)
        .get(entityId, stepKey, event);
      if (done) return undefined;
      const cur = this.db.prepare(`SELECT state FROM entities WHERE entity_id = ?`).get(entityId) as { state: string } | undefined;
      const from = cur?.state ?? machine.def.initial;
      const to = machine.next(from, event);
      this.db
        .prepare(`INSERT INTO entities (entity_id, kind, state) VALUES (?, ?, ?) ON CONFLICT (entity_id) DO UPDATE SET state = excluded.state`)
        .run(entityId, kind, to);
      this.db
        .prepare(`INSERT INTO entity_transitions (entity_id, step_key, event, from_state, to_state) VALUES (?, ?, ?, ?, ?)`)
        .run(entityId, stepKey, event, from, to);
      return to;
    });
  }

  entityState(entityId: string): string | undefined {
    return (this.db.prepare(`SELECT state FROM entities WHERE entity_id = ?`).get(entityId) as { state: string } | undefined)?.state;
  }

  saveContract(agentId: string, version: string, hash: string, shape: unknown): void {
    this.db
      .prepare(`INSERT INTO contracts (agent_id, version, hash, shape) VALUES (?, ?, ?, ?) ON CONFLICT (agent_id, version) DO UPDATE SET hash = excluded.hash, shape = excluded.shape`)
      .run(agentId, version, hash, canonicalJson(shape));
  }

  /** Test and audit helper: a whole table as canonical text, in a fixed order. */
  dump(table: 'envelopes' | 'entities' | 'steps', orderBy: string): string {
    if (!['envelopes', 'entities', 'steps'].includes(table) || !/^[a-z_]+(, ?[a-z_]+)*$/.test(orderBy)) {
      throw new Error('dump: refused');
    }
    const rows = this.db.prepare(`SELECT * FROM ${table} ORDER BY ${orderBy}`).all();
    return canonicalJson(rows);
  }
}
```

```ts
import { GENESIS_HASH, type Store, eventHash } from './store.js';

export type AuditResult = { ok: true; events: number } | { ok: false; brokenAt: number; reason: string };

/**
 * Walks the event chain from genesis, then checks it against the stored chain head.
 * Edited, inserted, deleted and truncated events are all caught.
 *
 * Limits: the chain is unkeyed, so it detects accidental and naive tampering only. A writer
 * with full database access can rewrite events and the head consistently; pass `expectedHead`
 * (an out-of-band copy, e.g. noted by the user) to catch that. Envelope bodies and the cost
 * ledger are not chained.
 */
export function verifyChain(store: Store, expectedHead?: { hash: string; count: number }): AuditResult {
  let prev = GENESIS_HASH;
  const rows = store.events();
  for (const r of rows) {
    if (r.prev_hash !== prev) return { ok: false, brokenAt: r.seq, reason: 'prev_hash does not match the previous event' };
    if (eventHash(prev, r) !== r.hash) return { ok: false, brokenAt: r.seq, reason: 'hash does not match the event contents' };
    prev = r.hash;
  }
  const head = store.chainHead();
  if (rows.length > 0 || head !== undefined) {
    if (head === undefined || head.count !== rows.length || head.hash !== prev) {
      return {
        ok: false,
        brokenAt: rows[rows.length - 1]?.seq ?? 0,
        reason: 'chain head does not match the events (truncated or rewritten tail)',
      };
    }
  }
  if (expectedHead !== undefined && (head === undefined || head.hash !== expectedHead.hash || head.count !== expectedHead.count)) {
    return { ok: false, brokenAt: rows[rows.length - 1]?.seq ?? 0, reason: 'chain head differs from the expected head' };
  }
  return { ok: true, events: rows.length };
}
```

```ts
import { hashOf, sha256 } from './canonical.js';
import { type AnyContract, contractMajor } from './contract.js';
import type { Envelope } from './envelope.js';

/**
 * Same agent major, same input, same handoff, same policy → same key → the
 * step never runs twice. The input envelope id is deliberately excluded:
 * identical content arriving by two routes is one piece of work.
 */
export function computeStepKey(contract: AnyContract, input: Envelope, policyHash: string): string {
  return hashOf({
    agent: `${contract.id}@${contractMajor(contract)}`,
    inputType: input.type,
    input: input.payload,
    handoff: { guarantees: input.handoff.guarantees, dataClass: input.handoff.dataClass, obligations: input.handoff.obligations },
    policyHash,
  });
}

/** Retry delay with jitter derived from the step key, so retry timing is reproducible too. */
export function backoffDelay(stepKey: string, attempt: number, backoffMs: readonly number[]): number {
  const base = backoffMs[attempt] ?? backoffMs.at(-1) ?? 0;
  return base + (Number.parseInt(sha256(`${stepKey}:${attempt}`).slice(0, 4), 16) % 250);
}
```

```ts
import type { AnyAgent } from './agent.js';
import { issueContext } from './broker.js';
import type { Clock } from './clock.js';
import { contractHash } from './contract.js';
import { maxClass } from './data-class.js';
import { type Cost, type Envelope, type GuardrailVerdict, makeEnvelopeId, nextObligations, ZERO_COST } from './envelope.js';
import { classify, type ErrorCode, KernelError } from './errors.js';
import type { GuardrailRegistry } from './guardrails/index.js';
import { backoffDelay, computeStepKey } from './memo.js';
import type { Policy } from './policy.js';
import type { Providers } from './providers.js';
import type { Store } from './store/store.js';

export interface ExecutorDeps {
  readonly store: Store;
  readonly policy: Policy;
  readonly policyHash: string;
  readonly providers: Providers;
  readonly clock: Clock;
  readonly guardrails: GuardrailRegistry;
  readonly contractHashOf: (agentId: string, version: string) => string | undefined;
  readonly sleep: (ms: number) => Promise<void>;
  /** Called right before agent.run; tests count real invocations with it. */
  readonly onInvoke?: (agentId: string, stepKey: string) => void;
}

export type StepOutcome =
  | { readonly kind: 'done'; readonly stepKey: string; readonly outputs: readonly Envelope[]; readonly replayed: boolean }
  | { readonly kind: 'quarantined'; readonly stepKey: string; readonly verdicts: readonly GuardrailVerdict[] }
  | { readonly kind: 'failed'; readonly stepKey: string; readonly error: ErrorCode; readonly detail: string }
  | { readonly kind: 'skipped'; readonly stepKey: string; readonly reason: 'budget' };

/** Rejects when the signal aborts, even if the agent ignores the signal. */
function untilAborted<T>(p: Promise<T>, signal: AbortSignal): Promise<T> {
  if (signal.aborted) return Promise.reject(signal.reason);
  return new Promise<T>((resolve, reject) => {
    const onAbort = () => reject(signal.reason);
    signal.addEventListener('abort', onAbort, { once: true });
    p.then(
      (v) => {
        signal.removeEventListener('abort', onAbort);
        resolve(v);
      },
      (e: unknown) => {
        signal.removeEventListener('abort', onAbort);
        reject(e);
      },
    );
  });
}

const addCost = (a: Cost, b: Cost): Cost => ({
  usd: a.usd + b.usd,
  inputTokens: a.inputTokens + b.inputTokens,
  outputTokens: a.outputTokens + b.outputTokens,
});

const messageOf = (err: unknown): string => {
  try {
    return err instanceof Error ? err.message : String(err);
  } catch {
    return 'unprintable error';
  }
};

/**
 * Spec §6.1: the same thirteen steps for every agent. Never throws: anything
 * unexpected becomes a failed step, so the supervisor always gets an outcome.
 */
export async function executeStep(deps: ExecutorDeps, runId: string, agent: AnyAgent, input: Envelope): Promise<StepOutcome> {
  const c = agent.contract; // 1 contract lookup (registry.get already threw if unknown)
  const stepKey = computeStepKey(c, input, deps.policyHash); // 5
  /** Every cent paid in this step, summed over all attempts. */
  let stepSpent: Cost = ZERO_COST;
  const eventOf = (kind: string, data: Record<string, unknown>) => ({ at: deps.clock.now(), kind, data: { agent: c.id, stepKey, ...data }, policyHash: deps.policyHash });
  /** A standalone event, for outcomes that write no step row. */
  const event = (kind: string, data: Record<string, unknown>): void => {
    const e = eventOf(kind, data);
    deps.store.appendEvent(runId, e.at, e.kind, e.data, e.policyHash);
  };
  /** Step row, envelopes, cost and the hash-chained event in one transaction. */
  const finish = (
    status: 'done' | 'failed' | 'quarantined',
    envelopes: readonly Envelope[],
    ev: ReturnType<typeof eventOf>,
    more: { errorCode?: ErrorCode; detail?: string } = {},
  ): void => deps.store.finishStep({ runId, stepKey, agentId: c.id, status, envelopes, cost: stepSpent, ...more, event: ev });
  const fail = (error: ErrorCode, detail: string): StepOutcome => {
    finish('failed', [], eventOf('step_failed', { error, detail }), { errorCode: error, detail });
    return { kind: 'failed', stepKey, error, detail };
  };

  try {
    const gctx = { contract: c, policy: deps.policy, policyHash: deps.policyHash, contractHashOf: deps.contractHashOf, input };
    const hash = contractHash(c);
    const running = { stepKey, agentId: c.id, version: c.version, contractHash: hash, inputEnvelopeId: input.envelopeId };

    // 2-4 lineage, preconditions and input guardrails. A trip is a property of
    // this envelope, not of the work, so it writes no step row and is never replayed.
    const gin = deps.guardrails.run('in', { ...gctx, side: 'in' });
    if (gin.tripped) {
      event('step_quarantined', { side: 'in', verdict: gin.verdicts.at(-1) });
      return { kind: 'quarantined', stepKey, verdicts: gin.verdicts };
    }

    // 6 replay: a finished step is never executed again
    const prior = deps.store.getStep(stepKey);
    if (prior?.status === 'done') {
      event('step_replayed', { status: 'done' });
      return { kind: 'done', stepKey, outputs: deps.store.envelopesOfStep(stepKey), replayed: true };
    }
    if (prior?.status === 'quarantined') {
      event('step_replayed', { status: 'quarantined' });
      return { kind: 'quarantined', stepKey, verdicts: JSON.parse(prior.detail ?? '[]') as GuardrailVerdict[] };
    }

    // 7 run budget (per agent and per run), counting what this step has already paid
    const overBudget = (): boolean =>
      (c.budget.usdPerRun > 0 || c.kind === 'llm') &&
      (deps.store.runCost(runId, c.id) + stepSpent.usd >= c.budget.usdPerRun ||
        deps.store.runCost(runId) + stepSpent.usd >= deps.policy.maxUsdPerRun);
    if (overBudget()) {
      event('step_skipped', { reason: 'budget' });
      return { kind: 'skipped', stepKey, reason: 'budget' };
    }

    // 8-9 capability-scoped context, timeout, bounded retries
    for (let attempt = 0; ; attempt++) {
      if (attempt > 0 && overBudget()) return fail('budget_exceeded', 'G-BUDGET: retry would exceed the run budget');
      deps.store.markRunning(running);
      const signal = AbortSignal.timeout(c.timeoutMs);
      const ctx = issueContext({ contract: c, policy: deps.policy, inputClass: input.handoff.dataClass, providers: deps.providers, clock: deps.clock, signal });
      let result: { ok: true; items: readonly unknown[] } | { ok: false; err: unknown };
      try {
        deps.onInvoke?.(c.id, stepKey);
        const items: unknown = await untilAborted(agent.run(input.payload, ctx), signal);
        if (!Array.isArray(items)) throw new KernelError('invalid_output', 'agent.run must return an array');
        result = { ok: true, items };
      } catch (err) {
        result = { ok: false, err };
      }
      stepSpent = addCost(stepSpent, ctx.spent());

      if (!result.ok) {
        const code = classify(result.err);
        if (c.retry.on.includes(code) && attempt < c.retry.max) {
          event('step_retry', { attempt: attempt + 1, error: code });
          await deps.sleep(backoffDelay(stepKey, attempt, c.retry.backoffMs));
          continue;
        }
        return fail(code, messageOf(result.err));
      }

      // 10-11 schema, data class, postconditions on every output
      // Taint: outputs inherit the input's class. Only a declared redactor sets its own (G-DATACLASS checks it).
      const dataClass = c.redactor ? maxClass(c.dataClasses.out) : input.handoff.dataClass;
      const outputs: Envelope[] = [];
      for (const [i, payload] of result.items.entries()) {
        const gout = deps.guardrails.run('out', { ...gctx, side: 'out', output: { payload, dataClass } });
        if (gout.tripped) {
          finish('quarantined', [], eventOf('step_quarantined', { side: 'out', index: i, verdict: gout.verdicts.at(-1) }), { detail: JSON.stringify(gout.verdicts) });
          return { kind: 'quarantined', stepKey, verdicts: gout.verdicts };
        }
        outputs.push({
          envelopeId: makeEnvelopeId(stepKey, i),
          stepKey,
          index: i,
          type: c.produces.id,
          producer: { agentId: c.id, version: c.version, contractHash: hash },
          parents: [input.envelopeId],
          payload,
          handoff: {
            guarantees: c.guarantees.map((g) => g.id).sort(),
            dataClass,
            obligations: nextObligations(input.handoff.obligations, c),
            pins: { policyHash: deps.policyHash },
          },
          guardrails: [...gin.verdicts, ...gout.verdicts],
          cost: i === 0 ? stepSpent : ZERO_COST,
        });
      }

      // 12-13 one transaction: envelopes, step status, cost and the hash-chained event
      finish('done', outputs, eventOf('step_done', { outputs: outputs.length, usd: stepSpent.usd }));
      return { kind: 'done', stepKey, outputs, replayed: false };
    }
  } catch (err) {
    // Unexpected: a store error, a rejecting sleep, a refused finish. Record what we can, then report.
    const detail = messageOf(err);
    let recorded = false;
    try {
      if (deps.store.getStep(stepKey)?.status === 'running') {
        fail('unknown', detail);
        recorded = true;
      }
    } catch {
      // best effort: the store itself may be what failed
    }
    if (!recorded) {
      try {
        event('step_failed', { error: 'unknown', detail });
      } catch {
        // best effort
      }
    }
    return { kind: 'failed', stepKey, error: 'unknown', detail };
  }
}
```

```ts
import { Breakers } from './breaker.js';
import { canonicalJson, hashOf } from './canonical.js';
import type { Clock } from './clock.js';
import { contractHash, contractShape } from './contract.js';
import { type Envelope, makeEnvelopeId, openObligations, ZERO_COST } from './envelope.js';
import { InvalidTransition } from './errors.js';
import { executeStep, type ExecutorDeps, type StepOutcome } from './executor.js';
import type { GuardrailRegistry } from './guardrails/index.js';
import type { Policy } from './policy.js';
import type { Providers } from './providers.js';
import type { Registry } from './registry.js';
import type { StateMachine } from './state-machine.js';
import type { EventRow, Store } from './store/store.js';
import { type CompiledPipeline, SUPERVISOR_ID } from './wiring.js';

/** Maps envelopes and outcomes onto an entity state machine (e.g. one row per job). */
export interface Tracker {
  readonly kind: string;
  readonly machine: StateMachine;
  keyOf(env: Envelope): string | null;
  eventFor(agentId: string, outcome: 'done' | 'quarantined'): string | null;
}

export interface SupervisorDeps {
  readonly store: Store;
  readonly registry: Registry;
  readonly compiled: CompiledPipeline;
  readonly guardrails: GuardrailRegistry;
  readonly policy: Policy;
  readonly policyHash: string;
  readonly providers: Providers;
  readonly clock: Clock;
  readonly sleep?: (ms: number) => Promise<void>;
  readonly tracker?: Tracker;
  readonly onInvoke?: (agentId: string, stepKey: string) => void;
  /** Test hook: runs after a step's own transaction, before routing. Throwing here simulates a crash. */
  readonly afterStep?: (count: number) => void;
}

export interface RunReport {
  readonly runId: string;
  readonly status: 'completed';
  readonly queue: Record<'pending' | 'done' | 'failed' | 'quarantined' | 'skipped', number>;
  readonly replayed: number;
  readonly openObligations: readonly { readonly envelopeId: string; readonly ids: readonly string[] }[];
  readonly costUsd: number;
}

/**
 * The deterministic hub. It never calls a model and never decides anything a
 * table doesn't: routes come from the compiled wiring, order from the queue's
 * total order, state from the tracker's machine.
 */
export class Supervisor {
  /**
   * Breakers are per run and in memory only: their counts restart after a
   * resume, so a resumed run re-learns failures. Everything else a report
   * needs is read back from the store.
   */
  private readonly breakersByRun = new Map<string, Breakers>();

  constructor(private readonly d: SupervisorDeps) {}

  private get exec(): ExecutorDeps {
    const d = this.d;
    return {
      store: d.store,
      policy: d.policy,
      policyHash: d.policyHash,
      providers: d.providers,
      clock: d.clock,
      guardrails: d.guardrails,
      sleep: d.sleep ?? ((ms) => new Promise((r) => setTimeout(r, ms))),
      contractHashOf: (id, v) => (id === SUPERVISOR_ID && v === d.compiled.pipeline.version ? d.compiled.seedHash : d.registry.contractHashOf(id, v)),
      ...(d.onInvoke ? { onInvoke: d.onInvoke } : {}),
    };
  }

  private breakers(runId: string): Breakers {
    let b = this.breakersByRun.get(runId);
    if (!b) this.breakersByRun.set(runId, (b = new Breakers()));
    return b;
  }

  private event(runId: string, kind: string, data: unknown): void {
    this.d.store.appendEvent(runId, this.d.clock.now(), kind, data, this.d.policyHash);
  }

  /** Validates seeds, records the run and drains its queue. */
  async start(runId: string, seeds: readonly unknown[]): Promise<RunReport> {
    const { store, compiled, policyHash, clock } = this.d;
    if (store.getRun(runId)) throw new Error(`run ${runId} already exists; resume it instead`);
    const seedType = compiled.pipeline.seed.type;
    const envs: Envelope[] = seeds.map((payload) => {
      const r = seedType.schema.safeParse(payload);
      if (!r.success || canonicalJson(r.data) !== canonicalJson(payload)) throw new TypeError(`seed is not a valid ${seedType.id}`);
      for (const g of compiled.pipeline.seed.guarantees) if (!g.test(payload)) throw new TypeError(`seed fails ${g.id}`);
      const stepKey = hashOf({ seed: compiled.pipeline.id, version: compiled.pipeline.version, payload, policyHash });
      return {
        envelopeId: makeEnvelopeId(stepKey, 0),
        stepKey,
        index: 0,
        type: seedType.id,
        producer: { agentId: SUPERVISOR_ID, version: compiled.pipeline.version, contractHash: compiled.seedHash },
        parents: [],
        payload,
        handoff: { guarantees: compiled.pipeline.seed.guarantees.map((g) => g.id).sort(), dataClass: 'P0', obligations: [], pins: { policyHash } },
        guardrails: [],
        cost: ZERO_COST,
      };
    });
    store.tx(() => {
      store.startRun(runId, `${compiled.pipeline.id}@${compiled.pipeline.version}`, policyHash, clock.now());
      for (const a of this.d.registry.list()) store.saveContract(a.contract.id, a.contract.version, contractHash(a.contract), contractShape(a.contract));
      for (const e of envs) {
        store.putSeed(e.stepKey, e, compiled.pipeline.version, compiled.seedHash);
        this.route(runId, e);
      }
      this.event(runId, 'run_started', { pipeline: compiled.pipeline.id, seeds: envs.length });
    });
    return this.drain(runId);
  }

  /** Continues a crashed run. Finished steps replay from the store; nothing finished runs again. */
  async resume(runId: string): Promise<RunReport> {
    const run = this.d.store.getRun(runId);
    if (!run) throw new Error(`unknown run ${runId}`);
    if (run.status === 'completed') throw new Error(`run ${runId} is already completed`);
    if (run.policy_hash !== this.d.policyHash) throw new Error(`run ${runId} used a different policy; start a new run instead`);
    this.event(runId, 'run_resumed', {});
    return this.drain(runId);
  }

  private route(runId: string, env: Envelope): void {
    const consumers = this.d.compiled.routes.get(env.type) ?? [];
    for (const to of consumers) this.d.store.enqueue(runId, this.d.compiled.stage.get(to) ?? 0, to, env.envelopeId);
    if (consumers.length === 0) {
      const ids = openObligations(env);
      if (ids.length > 0) this.event(runId, 'obligation_open', { envelopeId: env.envelopeId, ids });
    }
  }

  private failure(key: string, threshold: number, runId: string): void {
    if (this.breakers(runId).failure(key, threshold)) this.event(runId, 'breaker_open', { key });
  }

  private async drain(runId: string): Promise<RunReport> {
    const { store, registry, compiled } = this.d;
    const breakers = this.breakers(runId);
    let count = 0;
    for (let item = store.nextQueued(runId); item; item = store.nextQueued(runId)) {
      const agent = registry.get(item.agent_id);
      const team = compiled.teamOf.get(item.agent_id);
      const agentKey = `agent:${item.agent_id}`;
      // An agent without a team has no team breaker.
      const teamKey = team ? `team:${team.id}` : undefined;
      if (breakers.isOpen(agentKey) || (teamKey !== undefined && breakers.isOpen(teamKey))) {
        store.markQueue(runId, item.agent_id, item.envelope_id, 'skipped');
        this.event(runId, 'step_skipped', { agent: item.agent_id, reason: 'breaker_open' });
        continue;
      }
      const input = store.getEnvelope(item.envelope_id);
      const outcome: StepOutcome = await executeStep(this.exec, runId, agent, input);
      count += 1;
      this.d.afterStep?.(count);

      store.tx(() => {
        if (outcome.kind === 'done') {
          for (const out of outcome.outputs) {
            this.route(runId, out);
            this.track(runId, out, outcome.stepKey, item.agent_id, 'done');
          }
        } else if (outcome.kind === 'quarantined') {
          this.track(runId, input, outcome.stepKey, item.agent_id, 'quarantined');
        }
        store.markQueue(runId, item.agent_id, item.envelope_id, outcome.kind);
      });

      if (outcome.kind === 'failed') {
        this.failure(agentKey, agent.contract.breaker.consecutiveFailures, runId);
        if (team && teamKey !== undefined) this.failure(teamKey, team.breaker.consecutiveFailures, runId);
      } else if (outcome.kind === 'done') {
        breakers.success(agentKey);
        if (teamKey !== undefined) breakers.success(teamKey);
      }
    }
    store.finishRun(runId, this.d.clock.now());
    const events = store.events(runId);
    const report: RunReport = {
      runId,
      status: 'completed',
      queue: store.queueCounts(runId),
      replayed: events.filter((e) => e.kind === 'step_replayed').length,
      openObligations: openObligationsOf(events),
      costUsd: store.runCost(runId),
    };
    this.event(runId, 'run_finished', report);
    return report;
  }

  private track(runId: string, env: Envelope, stepKey: string, agentId: string, outcome: 'done' | 'quarantined'): void {
    const t = this.d.tracker;
    if (!t) return;
    const key = t.keyOf(env);
    const ev = t.eventFor(agentId, outcome);
    if (key === null || ev === null) return;
    try {
      this.d.store.applyTransition(key, t.kind, stepKey, ev, t.machine);
    } catch (err) {
      // The same entity can legitimately arrive twice (two sources list the same job): it keeps its state and the duplicate is audited.
      if (!(err instanceof InvalidTransition)) throw err;
      this.event(runId, 'transition_rejected', { entity: key, event: ev, agent: agentId, stepKey });
    }
  }
}

/** The run's `obligation_open` events, one per envelope, in envelope order. */
function openObligationsOf(events: readonly EventRow[]): RunReport['openObligations'] {
  const byEnvelope = new Map<string, readonly string[]>();
  for (const e of events) {
    if (e.kind !== 'obligation_open') continue;
    const d = JSON.parse(e.data) as { envelopeId: string; ids: string[] };
    byEnvelope.set(d.envelopeId, d.ids);
  }
  return [...byEnvelope]
    .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0))
    .map(([envelopeId, ids]) => ({ envelopeId, ids }));
}
```

```ts
import { existsSync, mkdirSync, readdirSync, readFileSync, statSync, unlinkSync, writeFileSync } from 'node:fs';
import { basename, join } from 'node:path';
import { canonicalJson } from './canonical.js';
import { contractHash, contractMajor, contractShape } from './contract.js';
import type { Registry } from './registry.js';

export interface ContractDoc {
  readonly hash: string;
  readonly shape: Record<string, unknown>;
}

const fileOf = (dir: string, id: string, major: string) => join(dir, `${id}@${major}.json`);

/** Pretty, key-sorted JSON so diffs are reviewable. */
function render(doc: ContractDoc): string {
  return `${JSON.stringify(JSON.parse(canonicalJson(doc)), null, 2)}\n`;
}

export function exportContracts(registry: Registry, dir: string): string[] {
  mkdirSync(dir, { recursive: true });
  const paths = registry.list().map((a) => {
    const c = a.contract;
    const path = fileOf(dir, c.id, contractMajor(c));
    writeFileSync(path, render({ hash: contractHash(c), shape: contractShape(c) }));
    return path;
  });
  // The directory is generated: anything this run did not write is stale.
  const written = new Set(paths.map((p) => basename(p)));
  for (const f of readdirSync(dir)) if (f.endsWith('.json') && !written.has(f) && statSync(join(dir, f)).isFile()) unlinkSync(join(dir, f));
  return paths;
}

const semverParts = (v: string): number[] => v.split('.').map((n) => Number.parseInt(n, 10) || 0);

/** Numeric x.y.z comparison: negative when a < b. */
function compareVersions(a: string, b: string): number {
  const [pa, pb] = [semverParts(a), semverParts(b)];
  for (let i = 0; i < 3; i++) {
    const d = (pa[i] ?? 0) - (pb[i] ?? 0);
    if (d !== 0) return d;
  }
  return 0;
}

function readDoc(path: string): ContractDoc | undefined {
  try {
    const doc = JSON.parse(readFileSync(path, 'utf8')) as Partial<ContractDoc> | null;
    if (doc && typeof doc.hash === 'string' && typeof doc.shape === 'object' && doc.shape !== null) return doc as ContractDoc;
  } catch {
    // fall through: unreadable
  }
  return undefined;
}

/** A contract that changed shape without a version bump is drift: other agents rely on the old shape. */
export function checkDrift(registry: Registry, dir: string): string[] {
  const problems: string[] = [];
  for (const a of registry.list()) {
    const c = a.contract;
    const path = fileOf(dir, c.id, contractMajor(c));
    if (!existsSync(path)) {
      problems.push(`${c.id}@${contractMajor(c)}: no exported contract; run node --run contracts:export`);
      continue;
    }
    const doc = readDoc(path);
    if (!doc) {
      problems.push(`${c.id}@${contractMajor(c)}: contract file is unreadable; run node --run contracts:export`);
      continue;
    }
    const committedVersion = (doc.shape as { version?: string }).version;
    if (committedVersion === c.version && doc.hash !== contractHash(c)) {
      problems.push(`${c.id}@${c.version}: contract changed without a version bump`);
    } else if (committedVersion === c.version && canonicalJson(doc.shape) !== canonicalJson(contractShape(c))) {
      problems.push(`${c.id}@${c.version}: contracts-out file was edited by hand; run node --run contracts:export`);
    } else if (committedVersion !== c.version && typeof committedVersion === 'string' && compareVersions(committedVersion, c.version) > 0) {
      problems.push(`${c.id}: code version ${c.version} is older than the exported ${committedVersion}`);
    } else if (committedVersion !== c.version) {
      problems.push(`${c.id}: version ${c.version} is not exported yet (committed ${committedVersion}); run node --run contracts:export`);
    }
  }
  const expected = new Set(registry.list().map((a) => `${a.contract.id}@${contractMajor(a.contract)}.json`));
  const stale = (existsSync(dir) ? readdirSync(dir) : []).filter((f) => f.endsWith('.json') && !expected.has(f)).sort();
  for (const f of stale) problems.push(`${f}: no registered agent exports this contract; delete it or restore the agent`);
  return problems;
}
```

```ts
import { canonicalJson } from '../kernel/canonical.js';

export function exactMatch(pairs: readonly { expected: unknown; actual: unknown }[]): number {
  if (pairs.length === 0) return 0;
  return pairs.filter((p) => canonicalJson(p.expected) === canonicalJson(p.actual)).length / pairs.length;
}

/** For a rule or classifier: positive = "the rule fired". */
export function precisionRecall(pairs: readonly { expected: boolean; actual: boolean }[]): { precision: number; recall: number } {
  const tp = pairs.filter((p) => p.expected && p.actual).length;
  const fp = pairs.filter((p) => !p.expected && p.actual).length;
  const fn = pairs.filter((p) => p.expected && !p.actual).length;
  return { precision: tp + fp === 0 ? 1 : tp / (tp + fp), recall: tp + fn === 0 ? 1 : tp / (tp + fn) };
}

/** Average ranks for ties, then Pearson on the ranks. */
export function spearman(xs: readonly number[], ys: readonly number[]): number {
  if (xs.length !== ys.length || xs.length < 2) throw new RangeError('spearman needs two equal-length series of at least 2');
  const rank = (v: readonly number[]) => {
    const order = v.map((x, i) => ({ x, i })).sort((a, b) => a.x - b.x);
    const r = new Array<number>(v.length);
    for (let i = 0; i < order.length; ) {
      let j = i;
      while (j + 1 < order.length && order[j + 1]!.x === order[i]!.x) j++;
      for (let k = i; k <= j; k++) r[order[k]!.i] = (i + j) / 2 + 1;
      i = j + 1;
    }
    return r;
  };
  const rx = rank(xs);
  const ry = rank(ys);
  const mean = (a: number[]) => a.reduce((s, x) => s + x, 0) / a.length;
  const mx = mean(rx);
  const my = mean(ry);
  let num = 0;
  let dx = 0;
  let dy = 0;
  for (let i = 0; i < rx.length; i++) {
    const a = rx[i]! - mx;
    const b = ry[i]! - my;
    num += a * b;
    dx += a * a;
    dy += b * b;
  }
  return dx === 0 || dy === 0 ? 0 : num / Math.sqrt(dx * dy);
}
```

```ts
import { readFileSync } from 'node:fs';
import type { AnyAgent } from '../kernel/agent.js';
import { issueContext } from '../kernel/broker.js';
import { canonicalJson } from '../kernel/canonical.js';
import { fixedClock } from '../kernel/clock.js';
import type { Policy } from '../kernel/policy.js';
import type { Providers } from '../kernel/providers.js';
import { exactMatch } from './metrics.js';

export interface EvalCase {
  readonly id: string;
  readonly input: unknown;
  readonly expected: unknown;
}

export interface EvalSpec {
  readonly agent: AnyAgent;
  readonly cases: readonly EvalCase[];
  /** Maps the agent's outputs to what the dataset's `expected` describes. */
  readonly project: (outputs: readonly unknown[]) => unknown;
  readonly metric: 'exactMatch';
  readonly min: number;
}

export interface EvalReport {
  readonly agentId: string;
  readonly version: string;
  readonly metric: 'exactMatch';
  readonly n: number;
  readonly score: number;
  readonly min: number;
  readonly pass: boolean;
  readonly failures: readonly { readonly id: string; readonly expected: unknown; readonly actual: unknown }[];
}

export function loadJsonl(path: string): EvalCase[] {
  return readFileSync(path, 'utf8')
    .split('\n')
    .filter((l) => l.trim() !== '')
    .map((l, i) => {
      const c = JSON.parse(l) as Partial<EvalCase>;
      if (typeof c.id !== 'string' || !('input' in c) || !('expected' in c)) throw new TypeError(`${path}:${i + 1}: needs id, input, expected`);
      return c as EvalCase;
    });
}

/** Runs each case through the agent with the same broker it gets in production; schema failures count as wrong. */
export async function runEval(spec: EvalSpec, env: { providers: Providers; policy: Policy }): Promise<EvalReport> {
  const c = spec.agent.contract;
  const pairs: { id: string; expected: unknown; actual: unknown }[] = [];
  for (const k of spec.cases) {
    let actual: unknown;
    const input = c.consumes.schema.safeParse(k.input);
    if (!input.success) {
      actual = { error: 'input does not match the consumed schema' };
    } else {
      try {
        const ctx = issueContext({ contract: c, policy: env.policy, inputClass: 'P0', providers: env.providers, clock: fixedClock('2026-01-01T00:00:00Z'), signal: AbortSignal.timeout(c.timeoutMs) });
        const outs = await spec.agent.run(k.input, ctx);
        const bad = outs.find((o) => !c.produces.schema.safeParse(o).success);
        actual = bad === undefined ? spec.project(outs) : { error: 'output does not match the produced schema' };
      } catch (e) {
        actual = { error: (e as Error).message };
      }
    }
    pairs.push({ id: k.id, expected: k.expected, actual });
  }
  const score = exactMatch(pairs);
  return {
    agentId: c.id,
    version: c.version,
    metric: spec.metric,
    n: pairs.length,
    score,
    min: spec.min,
    pass: score >= spec.min,
    failures: pairs.filter((p) => canonicalJson(p.expected) !== canonicalJson(p.actual)),
  };
}

export function failedGates(reports: readonly EvalReport[]): string[] {
  return reports.filter((r) => !r.pass).map((r) => `${r.agentId}@${r.version}: ${r.metric} ${r.score.toFixed(3)} < ${r.min}`);
}
```
